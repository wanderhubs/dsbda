import pandas as pd
import numpy as np
from sklearn import datasets

iris = datasets.load_iris()

df = pd.DataFrame(iris.data, columns=iris.feature_names)

df['Species'] = iris.target

df['Species'] = df['Species'].map({
    0: 'Iris-setosa',
    1: 'Iris-versicolor',
    2: 'Iris-virginica'
})

df.columns = [
    'SepalLengthCm',
    'SepalWidthCm',
    'PetalLengthCm',
    'PetalWidthCm',
    'Species'
]

print("First 5 Rows:\n")
print(df.head())

print("\nShape of Dataset:")
print(df.shape)

print("\nOverall Summary Statistics:\n")
print(df.describe())

print("\nMean Values:\n")
print(df.mean(numeric_only=True))

print("\nMedian Values:\n")
print(df.median(numeric_only=True))

print("\nStandard Deviation:\n")
print(df.std(numeric_only=True))

print("\nMinimum Values:\n")
print(df.min(numeric_only=True))

print("\nMaximum Values:\n")
print(df.max(numeric_only=True))

print("\nGrouped Statistics by Species:\n")

grouped_stats = df.groupby('Species').agg(
    ['mean', 'median', 'min', 'max', 'std']
)

print(grouped_stats)

print("\nCount of Each Species:\n")
print(df['Species'].value_counts())

setosa = df[df['Species'] == 'Iris-setosa']

print("\nStatistics of Iris-setosa:\n")
print(setosa.describe())

versicolor = df[df['Species'] == 'Iris-versicolor']

print("\nStatistics of Iris-versicolor:\n")
print(versicolor.describe())

virginica = df[df['Species'] == 'Iris-virginica']

print("\nStatistics of Iris-virginica:\n")
print(virginica.describe())

print("\nPercentiles of Dataset:\n")

percentiles = df.describe(percentiles=[0.25, 0.50, 0.75])

print(percentiles)

print("\nVariance of Dataset:\n")
print(df.var(numeric_only=True))

# Measures of Central Tendency

Central tendency means:

👉 the center value of data

Main measures:

1. Mean
2. Median
3. Mode

---

# 1. Mean

Mean is the average of all values.

Formula:

\text{Mean} = \frac{\text{Sum of all values}}{\text{Number of values}}

Example:

Values:
10, 20, 30

Mean:

[
\frac{10+20+30}{3} = 20
]

### In Python

```python id="hixf5k"
df.mean()
```

---

# 2. Median

Median is the middle value after sorting data.

Example:

```text
10 20 30 40 50
```

Median = 30

If even numbers exist:

```text
10 20 30 40
```

Median:

[
\frac{20+30}{2} = 25
]

### In Python

```python id="ww9l2s"
df.median()
```

---

# 3. Mode

Mode is the value that appears most frequently.

Example:

```text
1 2 2 3 4
```

Mode = 2

### In Python

```python id="m3p2gi"
df.mode()
```

---

# Variance

Variance measures:
👉 how far values are spread from mean

Small variance:

* values are close to mean

Large variance:

* values are far from mean

Formula:

\text{Variance} = \frac{\sum (x-\bar{x})^2}{n}

### In Python

```python id="nlmowh"
df.var()
```

---

# Standard Deviation

Standard deviation is square root of variance.

It tells:
👉 how much data varies from average

Formula:

\sigma = \sqrt{\frac{\sum (x-\bar{x})^2}{n}}

Small SD:

* data close to mean

Large SD:

* data widely spread

### In Python

```python id="r5t0rd"
df.std()
```

---
