import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.datasets import load_iris

# Load Iris dataset
iris = load_iris()

# Create DataFrame
df = pd.DataFrame(iris.data, columns=iris.feature_names)

# Add target column
df['target'] = iris.target

# Display first 5 rows
print("First 5 rows of dataset:")
print(df.head())

# Dataset information
print("\nShape of Dataset:", df.shape)
print("\nColumns:", df.columns)
print("\nData Types:\n", df.dtypes)
print("\nBasic Statistics:\n", df.describe())

# Feature names only
features = iris.feature_names

# 1. Number of features and their types
print("\nNumber of Features:", len(features))

print("\nFeature Types:")
for col in features:
    print(col, "→ Numeric")

print("\nTarget Column:")
print("target → Nominal / Categorical")

# 2. Histogram for each feature
df[features].hist(figsize=(10, 8))
plt.suptitle("Histograms of Iris Features")
plt.show()

# 3. Boxplot for each feature
plt.figure(figsize=(10, 6))
sns.boxplot(data=df[features])
plt.title("Boxplot of Iris Features")
plt.show()

# Compare distributions using mean and median
print("\nComparison of Distributions:")

for col in features:
    print(f"\nFeature: {col}")
    print("Mean:", df[col].mean())
    print("Median:", df[col].median())

# Identify outliers using IQR method
print("\nOutlier Detection using IQR Method:")

for col in features:
    Q1 = df[col].quantile(0.25)
    Q3 = df[col].quantile(0.75)
    IQR = Q3 - Q1

    lower = Q1 - 1.5 * IQR
    upper = Q3 + 1.5 * IQR

    outliers = df[(df[col] < lower) | (df[col] > upper)]

    print(f"\nFeature: {col}")
    print("Outliers count:", len(outliers))

# Boxplot

---

# Outliers

Outliers are unusual or extreme values that are very different from other data values.

Outliers may occur because of:

* measurement errors
* incorrect data entry
* natural variation

In this practical, outliers are detected using the IQR method.

---

# IQR (Interquartile Range)

IQR is calculated as:

IQR = Q3 - Q1

Where:

* Q1 = First Quartile (25%)
* Q3 = Third Quartile (75%)

Outlier limits are calculated as:

Lower\ Limit = Q1 - 1.5 \times IQR

Upper\ Limit = Q3 + 1.5 \times IQR

Values outside these limits are considered outliers.
