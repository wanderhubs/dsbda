# Data Wrangling II

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.preprocessing import MinMaxScaler

# Load dataset
df = pd.read_csv("Final_Marks_Data.csv")

# Display first 5 rows
print(df.head())

# Dataset information
print(df.info())

# Check missing values
print(df.isnull().sum())

# Handle missing values
for col in df.columns:
    if df[col].dtype == 'object':
        df[col] = df[col].fillna(df[col].mode()[0])
    else:
        df[col] = df[col].fillna(df[col].median())

# Check duplicate rows
print("Duplicate rows:", df.duplicated().sum())

# Remove duplicates
df = df.drop_duplicates()

# Remove inconsistent values
df = df[(df['Attendance (%)'] >= 0) & (df['Attendance (%)'] <= 100)]
df = df[(df['Internal Test 1 (out of 40)'] >= 0) & (df['Internal Test 1 (out of 40)'] <= 40)]
df = df[(df['Internal Test 2 (out of 40)'] >= 0) & (df['Internal Test 2 (out of 40)'] <= 40)]
df = df[(df['Assignment Score (out of 10)'] >= 0) & (df['Assignment Score (out of 10)'] <= 10)]
df = df[(df['Daily Study Hours'] >= 0)]
df = df[(df['Final Exam Marks (out of 100)'] >= 0) & (df['Final Exam Marks (out of 100)'] <= 100)]

print("\nData after removing inconsistencies:")
print(df.describe())

# Select numeric columns
numeric_cols = df.select_dtypes(include=np.number).columns

# Boxplot for outlier visualization
for col in numeric_cols:
    plt.figure()
    df.boxplot(column=col)
    plt.title("Boxplot of " + col)
    plt.show()

# Detect and handle outliers using IQR + Capping Method
for col in numeric_cols:

    # Calculate Q1 and Q3
    Q1 = df[col].quantile(0.25)
    Q3 = df[col].quantile(0.75)

    # Calculate IQR
    IQR = Q3 - Q1

    # Find limits
    lower_limit = Q1 - 1.5 * IQR
    upper_limit = Q3 + 1.5 * IQR

    # Detect outliers
    outliers = df[(df[col] < lower_limit) | (df[col] > upper_limit)]

    print("\nColumn:", col)
    print("Number of outliers:", len(outliers))

    # Capping Method
    df[col] = np.where(df[col] < lower_limit, lower_limit, df[col])
    df[col] = np.where(df[col] > upper_limit, upper_limit, df[col])

print("\nOutliers handled successfully.")

# Data Transformation using Min-Max Scaling
scaler = MinMaxScaler()

df['Final Exam Marks Scaled'] = scaler.fit_transform(
    df[['Final Exam Marks (out of 100)']]
)

print("\nTransformed Data:")
print(df[['Final Exam Marks (out of 100)', 'Final Exam Marks Scaled']].head())

# Save cleaned dataset
df.to_csv("Cleaned_Final_Marks_Data.csv", index=False)

print("\nCleaned dataset saved successfully.")

Most common statistical method.

# Quartiles

| Quartile | Meaning        |
| -------- | -------------- |
| Q1       | 25% data point |
| Q3       | 75% data point |

# Finding Q1 and Q3

```python id="oljlwm"

Q3 = df[col].quantile(0.75)
```

---

# IQR Formula

IQR = Q3 - Q1

---

# Lower Limit Formula

\text{Lower Limit} = Q1 - 1.5 \times IQR

---

# Upper Limit Formula

\text{Upper Limit} = Q3 + 1.5 \times IQR

---
