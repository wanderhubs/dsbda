# Import seaborn library
# Seaborn is used for data visualization (graphs)
import seaborn as sns

# Import matplotlib library
# pyplot helps to display graphs
import matplotlib.pyplot as plt


# Load Titanic dataset
# Titanic dataset is already available inside seaborn
df = sns.load_dataset('titanic')


# Create graph size
# figsize = (width, height)
plt.figure(figsize=(6,4))


# Create Box Plot
sns.boxplot(

    # x-axis contains gender
    x='sex',

    # y-axis contains age
    y='age',

    # Different colors based on survival
    # 0 = Not Survived
    # 1 = Survived
    hue='survived',

    # Dataset name
    data=df
)


# Add graph title
plt.title("Box Plot: Age Distribution by Gender and Survival")


# Label for x-axis
plt.xlabel("Gender")


# Label for y-axis
plt.ylabel("Age")


# Display graph on screen
plt.show()


# -------------------------------------------------------
# Observations / Inference
# -------------------------------------------------------

# 1. Female passengers survived more than male passengers.

# 2. Most passengers were between 20 and 40 years old.

# 3. Many male passengers did not survive.

# 4. Females had better survival chances.

# 5. Some very old passengers are shown as outliers.

# What is a Box Plot?

A Box Plot is a statistical graph used to show the distribution of numerical data.

It displays:

* Minimum value
* Maximum value
* Median
* Quartiles
* Outliers

---

# Components of Box Plot

## 1. Minimum Value

Smallest value in dataset.

---

## 2. Maximum Value

Largest value in dataset.

---

## 3. Median

Middle value of data.

It divides data into two equal parts.

---

## 4. Quartiles

Quartiles divide data into four parts.

* Q1 → 25%
* Q2 → Median (50%)
* Q3 → 75%

---

## 5. Outliers

Values far away from normal data.

Example:

* Most ages = 20–40
* One age = 80
* 80 becomes outlier
